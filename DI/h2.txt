bubble sort and merge sort                                                                                                                                                                                                                                                                                               #include <iostream>
#include <omp.h>

using namespace std;

// ---------- Parallel Bubble Sort ----------
void bubbleSort(int a[], int n)
{
    for(int i = 0; i < n-1; i++)
    {
        #pragma omp parallel for
        for(int j = 0; j < n-i-1; j++)
        {
            if(a[j] > a[j+1])
                swap(a[j], a[j+1]);
        }
    }
}

// ---------- Merge Function ----------
void merge(int a[], int l, int m, int r)
{
    int temp[1000];
    int i = l, j = m+1, k = l;

    while(i <= m && j <= r)
    {
        if(a[i] < a[j])
            temp[k++] = a[i++];
        else
            temp[k++] = a[j++];
    }

    while(i <= m)
        temp[k++] = a[i++];

    while(j <= r)
        temp[k++] = a[j++];

    for(i = l; i <= r; i++)
        a[i] = temp[i];
}

// ---------- Parallel Merge Sort ----------
void mergeSort(int a[], int l, int r)
{
    if(l < r)
    {
        int m = (l + r) / 2;

        #pragma omp parallel sections
        {
            #pragma omp section
            mergeSort(a, l, m);

            #pragma omp section
            mergeSort(a, m+1, r);
        }

        merge(a, l, m, r);
    }
}

// ---------- Main ----------
int main()
{
    int n, a[1000], b[1000];

    cout << "Enter size: ";
    cin >> n;

    cout << "Enter elements:\n";

    for(int i = 0; i < n; i++)
    {
        cin >> a[i];
        b[i] = a[i];
    }

    // Bubble Sort
    double start = omp_get_wtime();

    bubbleSort(a, n);

    double end = omp_get_wtime();

    cout << "\nBubble Sort:\n";

    for(int i = 0; i < n; i++)
        cout << a[i] << " ";

    cout << "\nTime = " << end - start << " sec\n";

    // Merge Sort
    start = omp_get_wtime();

    mergeSort(b, 0, n-1);

    end = omp_get_wtime();

    cout << "\nMerge Sort:\n";

    for(int i = 0; i < n; i++)
        cout << b[i] << " ";

    cout << "\nTime = " << end - start << " sec\n";

    return 0;
}
